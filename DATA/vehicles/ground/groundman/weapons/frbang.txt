# first thing in the weapon file should always be the weapontype
# this causes the default_(weapontype).txt file to be loaded first
# valid weapon types are:
# SMG, SATCHEL, RIFLE, AUTO_PISTOL, GRENADE, REVOLVER, ITEM, KNIFE, LMG
weaponType = GRENADE

# specify the weapon & clip (clip optional) model here
# note that data\gr2data\FPS\ is prepended to these file names
weaponModel = frenchgrenade\frenchgrenade

# granny attach bone names for weapon & clip
# the weaponConnect bone in the weapon attaches to the weaponConnectTo bone in the skeleton
weaponConnectTo = weapon-control
weaponConnect = mills-connect

grenadeDelayTime = 450

# how far back along the barrel to start the bullet (firepoints are at the tip)
firepointDisplacement = 0.0

# this is in place to set a fire point outside the Granny model
# It will be used only if the firepoint is not in place
# This will be from line of sight of the eye point
# First we set the Z distance or the distance away from the eye point
firepointZDistance = 0.70

#now we set the Horizontal and Vertical offset for the fire point
firepointHorizOffset = 0.0
firepointVertOffset = 0.5

grenadeVisDelay = 0.5
